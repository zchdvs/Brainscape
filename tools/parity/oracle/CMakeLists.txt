cmake_minimum_required(VERSION 3.21)
project(brainscape_oracle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
function(variant name dsp)
  set(D ${CMAKE_CURRENT_SOURCE_DIR}/${dsp})
  add_library(dsp_${name} STATIC ${D}/src/Engine.cpp ${D}/src/Granular.cpp ${D}/src/OnsetDetector.cpp ${D}/src/PostChain.cpp)
  target_include_directories(dsp_${name} PUBLIC ${D}/include)
  target_compile_options(dsp_${name} PRIVATE ${ARGN})
  add_executable(x86_${name} oracle.cpp)
  target_link_libraries(x86_${name} PRIVATE dsp_${name})
endfunction()
if(MSVC)
  variant(msvc_orig      dsp_orig /GR- /fp:precise)
  variant(msvc_det       dsp_det  /GR- /fp:precise)
  variant(msvc_det_avx2  dsp_det  /GR- /fp:precise /arch:AVX2)
  variant(msvc_det_contract dsp_det /GR- /fp:precise /fp:contract /arch:AVX2)
else()
  variant(gcc_orig       dsp_orig -fno-rtti -fno-exceptions -ffp-contract=off)
  variant(gcc_det        dsp_det  -fno-rtti -fno-exceptions -ffp-contract=off)
  variant(gcc_det_v3     dsp_det  -fno-rtti -fno-exceptions -ffp-contract=off -march=x86-64-v3)
  variant(gcc_det_v3fma  dsp_det  -fno-rtti -fno-exceptions -march=x86-64-v3)
endif()
if(MSVC)
  variant(msvc_fma       dsp_fma  /GR- /fp:precise)
  variant(msvc_fma_avx2  dsp_fma  /GR- /fp:precise /arch:AVX2)
else()
  variant(gcc_fma        dsp_fma  -fno-rtti -fno-exceptions -ffp-contract=off)
  variant(gcc_fma_v3     dsp_fma  -fno-rtti -fno-exceptions -ffp-contract=off -march=x86-64-v3)
endif()
if(MSVC)
  variant(msvc_det_noftz dsp_det /GR- /fp:precise /DBRAINSCAPE_PROBE_NO_FZ /DBRAINSCAPE_ALLOW_NO_DENORMAL_GUARD)
endif()
